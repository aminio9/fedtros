Run these commands from the repository root in Bash, Git Bash, WSL, or Linux.
Copy the preamble once per shell session so tee cannot hide a failed Python process.

set -o pipefail
mkdir -p logs

================================================================================
DATA PREPARATION: Download and build all canonical CSV files PENDING
================================================================================

poetry run python scripts/prepare_external_datasets.py --dataset all \
  2>&1 | tee logs/prepare_external_datasets.log \
  && echo "DONE prepare_external_datasets" | tee -a run_status.txt \
  || echo "FAILED prepare_external_datasets" | tee -a run_status.txt

# Individual preparation alternatives:
poetry run python scripts/prepare_external_datasets.py --dataset btat
poetry run python scripts/prepare_external_datasets.py --dataset toniot
poetry run python scripts/prepare_external_datasets.py --dataset cicids2017

================================================================================
EXP1-BTAT: IID closed-set DKD-FedOS, all 7 classes, open-set disabled PENDING
================================================================================

poetry run python run.py experiment=exp1 \
  dataset=btat \
  +method=dkd_fedos \
  seed=42 \
  federated.num_clients=10 \
  federated.num_rounds=100 \
  training.local_episodes_per_round=10 \
  open_set.evt.backend=fed_digos \
  open_set.fed_digos.enabled=false \
  training.dkd_student_osr_enabled=false \
  training.dkd_student_open_set_enabled=false \
  training.generator.enabled=false \
  dataset.preprocessing.iid=true \
  tracking.run_id=e1_btat_iid_closed_dkd_fedos_seed42 \
  2>&1 | tee logs/e1_btat_iid_closed_dkd_fedos_seed42.log \
  && echo "DONE e1_btat_iid_closed_dkd_fedos_seed42" | tee -a run_status.txt \
  || echo "FAILED e1_btat_iid_closed_dkd_fedos_seed42" | tee -a run_status.txt
