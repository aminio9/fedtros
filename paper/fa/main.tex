\begin{abstract}
این مقاله چارچوب \CFMARLOS{} را برای تشخیص نفوذ در ترافیک \BNat{} تحت یادگیری فدرال افقی و داده‌های غیرهمسان ارائه می‌کند. سامانه پیشنهادی سه جزء را به‌صورت هم‌زمان ترکیب می‌کند: عامل محلی \CVAEDQN{} برای طبقه‌بندی کلاس‌های شناخته‌شده، لایه تجمیع فدرال برای حفظ حریم خصوصی و مقیاس‌پذیری، و واسنجی \EVT{} برای رد نمونه‌های ناشناخته. در این معماری، فقط پارامترهای یادگیریِ مشترک میان کلاینت‌ها فدرال می‌شوند؛ حافظه بازپخش، وضعیت بهینه‌ساز، داده خام و مصنوعات \EVT{} محلی یا پساآموزشی باقی می‌مانند. پس از آموزش فدرال، مدل جهانی با خطاهای بازسازیِ نمونه‌های اعتبارسنجیِ درست‌طبقه‌بندی‌شده واسنجی می‌شود تا بتواند حملات دیده‌نشده را با آستانه‌ای کنترل‌شده رد کند. این طراحی برای بهبود هم‌زمانِ دقت کلاس‌های شناخته‌شده، پایداری تحت ناهمگنی کلاینتی، و تشخیص مجموعه باز مناسب است.
\end{abstract}

\noindent\textbf{کلیدواژه‌ها:} یادگیری فدرال، تشخیص نفوذ، تشخیص مجموعه باز، \EVT، \DoubleDQN، ترافیک \BNat

\section{مقدمه}
ترافیک \BNat{} و سامانه‌های توزیع‌شده مشابه، به‌طور طبیعی در چندین گره، سازمان یا نقطه پایش جداگانه ذخیره می‌شوند. در چنین محیط‌هایی، متمرکز کردن داده خام همواره ممکن یا مطلوب نیست. یادگیری فدرال یک راه‌حل طبیعی فراهم می‌کند، اما قواعد تجمیع استاندارد مانند \FedAvg{} در حضور توزیع‌های غیرهمسان می‌توانند ناپایدار شوند. دشواری مسئله در تشخیص نفوذ بیشتر می‌شود، زیرا مدل نه‌تنها باید حملات شناخته‌شده را طبقه‌بندی کند، بلکه باید حملات دیده‌نشده را نیز به‌درستی رد کند.

پیاده‌سازی این پروژه یک خط لوله ترکیبی را دنبال می‌کند که در آن یادگیری بازنمایی نهفته، \DoubleDQN{}، مولد بازسازی، و \EVT{} در یک معماری فدرال همکارانه ادغام شده‌اند. در این مقاله، \CFMARLOS{} به‌عنوان یک روش یکپارچه برای طبقه‌بندی مجموعه بسته، رد ناشناخته و آموزش فدرال تحت ناهمگنی کلاینتی ارائه می‌شود.

\section{پیشینه و صورت مسئله}
این کار از دو منبع اصلی برای لایه تجمیع فدرال استفاده می‌کند. مقاله \LatinTerm{FMRL-LA} ایده ارتباط دو مرحله‌ای، منتقدهای ناهمگام، و تجمیع‌گر مرکزی برای انتخاب عامل‌های مفید را معرفی می‌کند \citep{zhang2024fmrl}. مقاله \LatinTerm{FedAWA} نیز نشان می‌دهد که بردار تغییرات مدل محلی یا client vector می‌تواند جهت به‌روزرسانی کلاینت را بدون نیاز به داده proxy آشکار کند و برای تنظیم وزن تجمیع به کار رود \citep{shi2025fedawa}. روش پیشنهادی این مقاله، \FMRLAVA{}، پیاده‌سازی مستقیم هیچ‌کدام از این دو روش نیست؛ فاز A از منطق انتخاب و critic در \LatinTerm{FMRL-LA} الهام گرفته، فاز B از هم‌راستایی بردار آپدیت در \LatinTerm{FedAWA} استفاده می‌کند، و هر دو بخش برای مسئله تشخیص نفوذ \CVAEDQN{}، رد مجموعه‌باز، و داده‌های غیرهمسان بازطراحی شده‌اند.

در این چارچوب، هر کلاینت مجموعه داده محلی \Di{} را در اختیار دارد و داده خام را با سرور به اشتراک نمی‌گذارد. هدف فدرال، کمینه‌سازی مجموع وزن‌دار زیان‌های محلی است:

\[
\min_{\Theta}\sum_{i=1}^{N}\frac{n_i}{\sum_j n_j}F_i(\Theta),
\]

که در آن \(n_i\) تعداد نمونه‌های محلی کلاینت \(i\) است. در حالت \FedProx{}، هدف محلی با جمله پروگزیمال به‌صورت زیر اصلاح می‌شود:

\[
F_i^{\mathrm{prox}}(\Theta)=F_i(\Theta)+\frac{\mu}{2}\lVert \Theta-\Theta^{k-1}\rVert_2^2.
\]

برای تشخیص مجموعه باز، به‌جای اتکا به اعتماد نرمِ مجموعه بسته، از خطای بازسازی و مدل‌سازی دم \EVT{} استفاده می‌شود. این کار به مدل اجازه می‌دهد تا نمونه‌هایی را که به توزیع کلاس‌های شناخته‌شده تعلق ندارند، با آستانه‌ای واسنجی‌شده رد کند.

\begin{algorithm}[t]
\caption{آموزش فدرال \CVAEDQN{} و واسنجی مجموعه‌باز}
\begin{algorithmic}[1]
\Require مجموعه داده‌های کلاینت‌ها $\{D_i\}_{i=1}^N$، مجموعه اعتبارسنجی $D_{\mathrm{val}}$، تعداد دورها $K$، پیکربندی محلی $H$، و راهبرد $S \in \{\LatinMath{FedAvg}, \LatinMath{FedProx}, \LatinMath{FMRL-AVA}\}$
\Ensure مدل نهایی $\theta^\ast$، مدل‌های \EVT{} $M_{\mathrm{evt}}$، و آستانه رد $\delta$
\State پارامترهای اولیه $\theta^0=\{\theta_p^0,\theta_r^0,\theta_Q^0,\theta_G^0\}$ را مقداردهی کن
\For{$k = 1$ تا $K$}
  \State سرور کلاینت‌های $C_k$ را نمونه‌گیری کرده و $\theta^{k-1}$ را به آن‌ها پخش می‌کند
  \ForAll{$i \in C_k$ \textbf{به‌صورت موازی}}
    \State $\mu_i \gets \mu$ اگر $S=\text{\LatinTerm{FedProx}}$ باشد، و در غیر این صورت $\mu_i \gets 0$
    \State $(\theta_i^k, m_i^k) \gets \LatinMath{ClientLocalUpdate}(D_i,\theta^{k-1},H,\mu_i)$
  \EndFor
  \If{$S=\text{\LatinTerm{FMRL-AVA}}$}
    \State $\theta^k \gets \LatinMath{FMRLAVAUpdate}(\theta^{k-1},\{(\theta_i^k,m_i^k)\})$
  \Else
    \State $\theta^k \gets \frac{\sum_{i\in C_k} n_i \theta_i^k}{\sum_{i\in C_k} n_i}$
  \EndIf
\EndFor
\State $\theta^\ast \gets \theta^K$
\State $(M_{\mathrm{evt}}, \delta) \gets \LatinMath{EVTCalibration}(D_{\mathrm{val}}, \theta^\ast)$
\State \Return $\theta^\ast, M_{\mathrm{evt}}, \delta$
\end{algorithmic}
\end{algorithm}

\subsection{عامل محلی \CVAEDQN}
فرض کنید \(s \in \mathbb{R}^d\) بردار ویژگی ترافیک و \(a \in \{1,\ldots,C\}\) یکی از کلاس‌های شناخته‌شده باشد. هر کلاینت مسئله را به‌صورت یک فرمول‌بندی تقویتیِ مبتنی بر مخزن داده مدل می‌کند؛ در این فرمول‌بندی، کنش همان کلاس پیش‌بینی‌شده است و پاداش با درستیِ پیش‌بینی تعیین می‌شود.

عامل محلی از چهار مؤلفه اصلی تشکیل شده است:
\begin{itemize}
  \item شبکه پیشین \(p_\theta(z \mid s)\) برای نگاشت حالت به توزیع نهفته؛
  \item شبکه بازشناسی \(q_\phi(z \mid s,a)\) برای استنتاج نهفته وابسته به کلاس؛
  \item شبکه \LatinTerm{Q} اصلی \(Q_\psi(z,s)\) برای انتخاب کلاس شناخته‌شده؛
  \item شبکه \LatinTerm{Q} هدف \(Q_{\psi^-}(z,s)\) برای پایدارسازی \DoubleDQN{}؛
  \item مولد \(G_\omega(z,a)\) برای بازسازی ویژگی‌ها.
\end{itemize}

شبکه پیشین با شبکه بازشناسی هم‌تراز می‌شود:

\[
L_{\mathrm{prior}}=
\mathrm{KL}\!\left(q_\phi(z \mid s,a^\ast)\,\|\,p_\theta(z \mid s)\right).
\]

برای به‌روزرسانی \LatinTerm{TD}، از هدف \DoubleDQN{} استفاده می‌شود:

\begin{align*}
a'&=\arg\max_a Q_\psi(z',s',a),\\
y&=r+\gamma(1-d)\,Q_{\psi^-}(z',s',a').
\end{align*}

\[
L_{\mathrm{TD}}=
\mathrm{Huber}\!\left(Q_\psi(z,s,a)-y\right).
\]

بازپخش تجربه و سیاست \(\epsilon\)-حریصانه برای یادگیری محلی استفاده می‌شوند. شبکه هدف فقط به‌صورت محلی همگام می‌شود و به‌عنوان یک مؤلفه فدرال مستقل ارسال نمی‌شود. اگر \FedProx{} فعال باشد، جمله پروگزیمال زیر نیز به هدف محلی افزوده می‌شود:

\[
L_{\mathrm{prox}}=
\frac{\mu}{2}\lVert \Theta_i-\Theta^{k-1}\rVert_2^2.
\]

\begin{algorithm}[t]
\caption{به‌روزرسانی محلی \CVAEDQN{} با گزینه \FedProx{}}
\begin{algorithmic}[1]
\Require داده محلی $D_i$، پارامترهای پخش‌شده $\theta^{k-1}$، پیکربندی محلی $H$، و ضریب پروگزیمال $\mu_i$
\Ensure پارامترهای به‌روزشده $\theta_i^k$ و شاخص‌های ممیزی $m_i^k$
\State $\theta^{k-1}$ را در $p_\theta$، $q_\phi$، $Q_\psi$ و $G_\omega$ بارگذاری کن
\State شبکه هدف را با $Q_\psi^- \gets Q_\psi$ مقداردهی کرده و $\theta^{k-1}$ را مرجع \FedProx{} قرار بده
\State حافظه بازپخش $B_i$ و سیاست $\epsilon$-حریصانه $\pi_i$ را مقداردهی کن
\For{هر اپیزود محلی}
  \State با محیط محلی تعامل کن و گذارهای $(s_t,a_t,r_t,s_{t+1},d_t,a_t^\ast)$ را در $B_i$ ذخیره کن
  \If{$|B_i|$ کافی باشد}
    \State یک مینی‌بچ از $B_i$ نمونه‌گیری کن
    \State $p_\theta$ را با $L_{\mathrm{prior}}$ و در صورت فعال بودن با جریمه \FedProx{} به‌روزرسانی کن
    \State $q_\phi$ و $Q_\psi$ را با $L_{\mathrm{TD}}$ و در صورت فعال بودن با جریمه \FedProx{} به‌روزرسانی کن
    \State $Q_\psi^-$ را به‌صورت نرم یا دوره‌ای از $Q_\psi$ همگام کن
  \EndIf
\EndFor
\State مجموعه نمونه‌های شناخته‌شده و درست‌طبقه‌بندی‌شده را بساز
\If{آموزش مولد فعال باشد}
  \State $G_\omega$ را فقط روی نمونه‌های شناخته‌شده و درست‌طبقه‌بندی‌شده آموزش بده
\EndIf
\State \Return $\theta_i^k=\{\theta_p,\theta_r,\theta_Q,\theta_G\}$ و $m_i^k$
\end{algorithmic}
\end{algorithm}

\subsection{بازسازی مبتنی بر مولد}
مولد پس از پیش‌بینی کلاس، نمونه را به‌صورت شرطی بازسازی می‌کند:

\[
\hat{x}=G_\omega(z,\hat{a}),
\qquad
e(x)=\frac{1}{d}\lVert x-\hat{x}\rVert_2^2 \cdot \kappa.
\]

مولد فقط روی نمونه‌های شناخته‌شده و درست‌طبقه‌بندی‌شده آموزش داده می‌شود. این انتخاب باعث می‌شود فضای بازسازی به منیفلد کلاس‌های شناخته‌شده نزدیک بماند و خطای بازسازی برای نمونه‌های ناشناخته، سیگنال معنادارتری داشته باشد.

\subsection{\FedAvg{} و \FedProx{} به‌عنوان مبنا}
\FedAvg{} از میانگین وزن‌دار بر اساس تعداد نمونه استفاده می‌کند:

\[
\Theta^k=\sum_{i \in C_k}\frac{n_i}{\sum_{j \in C_k}n_j}\Theta_i^k.
\]

\FedProx{} همان منطق سمت سرور را حفظ می‌کند، اما با جمله پروگزیمال، انحراف به‌روزرسانی‌های محلی را محدود می‌سازد. \FedAvg{} و \FedProx{} در این مقاله مبنا هستند و به‌عنوان مشارکت اصلی روش پیشنهادی تلقی نمی‌شوند.

\subsection{تجمیع همکارانه \FMRLAVA{}}
\FMRLAVA{} مؤلفه نوآورانه اصلی این کار است. این روش به‌جای فرض مساوی بودن تمام کلاینت‌ها، سودمندی هر کلاینت را پیش از تجمیع برآورد می‌کند و سپس یک ضریب هم‌راستایی بردار آپدیت را نیز وارد وزن نهایی می‌سازد. در فاز \LatinTerm{A}، کلاینت‌ها محلی آموزش می‌بینند و فراداده ممیزی شامل پاداش، \LatinTerm{F1}، دقت، پایداری \LatinTerm{TD}، نوآوری مبتنی بر \LatinTerm{KL}، آنتروپی، پوشش برچسب، کیفیت مولد و تعداد گام‌ها را ارسال می‌کنند. سرور این اطلاعات را با یک منتقد ناهمگام ارزیابی می‌کند و امتیاز سودمندی \(u_i\) را به دست می‌آورد.

در فاز \LatinTerm{B}، کلاینت‌های منتخب وزن‌های ذخیره‌شده را می‌فرستند و سرور دلتاهای پارامتری را ابتدا با وزن سودمندی و سپس با هم‌راستایی بردار تجمیع می‌کند:

\[
\Delta_i=\Theta_i^k-\Theta^{k-1},
\qquad
b_i=n_i u_i,
\qquad
\bar{\Delta}=\frac{\sum_{j \in A_k} b_j\Delta_j}{\sum_{j \in A_k} b_j},
\]

\[
m_i=\mathrm{clip}(\exp(\kappa \cos(\Delta_i,\bar{\Delta})),m_{\min},m_{\max}),
\qquad
a_i=b_i m_i,
\]

\[
\Theta^k=\Theta^{k-1}+\eta\frac{\sum_{i \in A_k} a_i\Delta_i}{\sum_{i \in A_k} a_i}.
\]

اگر داده‌ها \LatinTerm{IID} باشند و جهت آپدیت‌ها مشابه باشد، ضرایب \(m_i\) تقریباً به هم نزدیک می‌شوند و رفتار روش به \FedAvg{} نزدیک می‌ماند. در \LatinTerm{non-IID}، آپدیت‌هایی که با جهت مرجع دور ناسازگار هستند اثر کمتری می‌گیرند. هدف پایش سمت سرور همچنان یک \LatinTerm{reward} آگاه از اعتبارسنجی است، اما این \LatinTerm{reward} وزن مستقیم تجمیع را تعیین نمی‌کند و فقط برای آموزش و پایش منتقد و مخلوط‌کننده به‌کار می‌رود.

\begin{algorithm}[t]
\caption{انتخاب و تجمیع آگاه از سودمندی و هم‌راستایی بردار در \FMRLAVA{}}
\begin{algorithmic}[1]
\Require پارامترهای جهانی قبلی $\theta^{k-1}$، به‌روزرسانی‌ها و شاخص‌های کلاینتی $\{(\theta_i^k,m_i^k)\}$، آستانه سودمندی $\lambda$، حداقل کلاینت منتخب $q_{\min}$، حداکثر نسبت $\rho_{\max}$، نرخ گام $\eta$، شدت هم‌راستایی $\kappa$، و حدود $[m_{\min},m_{\max}]$
\Ensure پارامترهای جهانی به‌روزشده $\theta^k$
\State سودمندی $u_i$ را برای هر کلاینت با استفاده از خلاصه‌های نهفته و شاخص‌ها برآورد کن
\State مجموعه $A_k = \{i : u_i \ge \lambda\}$ را انتخاب کن
\If{$|A_k| < q_{\min}$}
  \State کلاینت‌های با بیشترین سودمندی را اضافه کن تا $|A_k| = q_{\min}$ شود
\EndIf
\State $|A_k| \le \lceil \rho_{\max}|C_k| \rceil$ را اعمال کن
\ForAll{$i \in A_k$}
  \State $\Delta_i \gets \theta_i^k - \theta^{k-1}$
\EndFor
\State $b_i \gets n_i u_i$ برای همه $i \in A_k$
\State $\bar{\Delta} \gets \frac{\sum_{i \in A_k} b_i\Delta_i}{\sum_{i \in A_k} b_i}$
\ForAll{$i \in A_k$}
  \State $m_i \gets \mathrm{clip}(\exp(\kappa \cos(\Delta_i,\bar{\Delta})),m_{\min},m_{\max})$
  \State $a_i \gets b_i m_i$
\EndFor
\State $\theta^k \gets \theta^{k-1} + \eta \frac{\sum_{i \in A_k} a_i\Delta_i}{\sum_{i \in A_k} a_i}$
\State منتقدهای سمت سرور و مخلوط‌کننده را با هدف پایشِ آگاه از اعتبارسنجی به‌روزرسانی کن
\State \Return $\theta^k$
\end{algorithmic}
\end{algorithm}

\subsection{واسنجی \EVT{} و استنتاج مجموعه باز}
پس از آموزش فدرال، مدل جهانی با \EVT{} واسنجی می‌شود. فقط نمونه‌های اعتبارسنجیِ شناخته‌شده و درست‌طبقه‌بندی‌شده برای برازش دم استفاده می‌شوند. برای هر کلاس، دم بالایی خطای بازسازی با توزیع پارتوی تعمیم‌یافته مدل می‌شود و سپس احتمال ناشناختگی محاسبه می‌گردد:

\[
P_{\mathrm{unknown}}(x)=M_{\mathrm{EVT},\hat{a}}(e(x)).
\]

آستانه جهانی \(\delta_{\mathrm{global}}\) از داده اعتبارسنجی تنظیم می‌شود. در زمان استنتاج، اگر

\[
P_{\mathrm{unknown}}(x)<\delta_{\mathrm{global}},
\]

نمونه به‌عنوان کلاس شناخته‌شده پذیرفته می‌شود؛ در غیر این صورت به‌عنوان ناشناخته رد می‌شود. این مرحله مدل را از طبقه‌بند مجموعه بسته به آشکارساز نفوذ مجموعه باز تبدیل می‌کند.

\begin{algorithm}[t]
\caption{واسنجی \EVT{} و استنتاج مجموعه باز}
\begin{algorithmic}[1]
\Require مجموعه اعتبارسنجی $D_{\mathrm{val}}$، مدل آموزش‌دیده $\theta^\ast$، کسر دُم $\alpha_{\mathrm{evt}}$، و نرخ مثبت کاذب هدف $\beta$
\Ensure مدل‌های \EVT{} $M_{\mathrm{evt}}$، آستانه جهانی $\delta$، و پیش‌بینی نهایی $\hat{y}$
\State خطاهای بازسازی نمونه‌های شناخته‌شده و درست‌طبقه‌بندی‌شده را جمع‌آوری کن
\ForAll{کلاس‌های شناخته‌شده $c$}
  \State یک مدل دُم \EVT{} را برای دُم بالایی خطاهای کلاس $c$ برازش کن
\EndFor
\State احتمالات ناشناختگی روی داده اعتبارسنجی را تخمین بزن و $\delta$ را از کمیتِ $(1-\beta)$ تعیین کن
\For{هر نمونه استنتاجی $x$}
  \State کلاس پیش‌بینی‌شده $\hat{c} = \arg\max_c Q_\psi(p_\theta(x),x,c)$ را محاسبه کن
  \State خطای بازسازی $e_x$ را با $q_\phi$ و $G_\omega$ محاسبه کن
  \State $p_u = M_{\mathrm{evt}}[\hat{c}](e_x)$ را محاسبه کن
  \If{$p_u \ge \delta$}
    \State $\hat{y} \gets$ ناشناخته
  \Else
    \State $\hat{y} \gets \hat{c}$
  \EndIf
\EndFor
\State \Return $M_{\mathrm{evt}}, \delta, \hat{y}$
\end{algorithmic}
\end{algorithm}

\subsection{تحلیل الگوریتم‌ها}
الگوریتم 1 جریان کامل آموزش فدرال و واسنجی نهایی را خلاصه می‌کند و نشان می‌دهد که آموزش فدرال و کالیبراسیون \EVT{} دو مرحله متمایز ولی پیوسته هستند. الگوریتم 2 به‌روزرسانی محلی \CVAEDQN{} را با یا بدون \FedProx{} مشخص می‌کند و روشن می‌سازد که شبکه هدف فقط نقش پایدارسازی محلی دارد. الگوریتم 3 قاعده انتخاب و تجمیع \FMRLAVA{} را توضیح می‌دهد و مزیت آن را در مواجهه با ناهمگنی کلاینتی نشان می‌دهد. الگوریتم 4 نیز برازش \EVT{} و تصمیم‌گیری مجموعه‌باز را در مرحله استنتاج ارائه می‌کند.

\subsection{منطق طراحی}
هر مؤلفه در \CFMARLOS{} یک شکست مشخص را هدف می‌گیرد: عامل \CVAEDQN{} رفتار طبقه‌بندی مجموعه بسته را می‌آموزد، مولد سیگنال بازسازی را فراهم می‌کند، لایه فدرال آموزش توزیع‌شده را ممکن می‌سازد، \FMRLAVA{} ناهمگنی کلاینتی را مدیریت می‌کند، و \EVT{} قانون ردِ واسنجی‌شده را برای حملات ناشناخته ارائه می‌دهد. این جداسازی وظایف، روش را برای آزمایش‌های حذف مؤلفه، تحلیل حساسیت، و تحلیل کارایی مناسب می‌سازد.

\section{پیکربندی تجربی}
پروتکل ارزیابی مطابق قرارداد مخزن تنظیم شده است. داده اصلی، ترافیک بلاکچین \BNat{} است. در تنظیم مجموعه‌باز، کلاس‌های «نرمال»، «بی‌پی»، «دی‌او‌اس» و «میتم» به‌عنوان کلاس‌های شناخته‌شده در نظر گرفته می‌شوند و «فوت» به‌عنوان حمله ناشناخته نگه داشته می‌شود. برای نسخه نهایی، تنظیم اصلی شامل 10 کلاینت و 100 دور منطقی فدرال است؛ تنظیمات کوچک‌تر فقط برای آزمون‌های سریع و توسعه در مخزن باقی مانده‌اند.

تمام مقایسه‌ها از همان قرارداد پیش‌پردازش، همان تقسیم آموزش/اعتبارسنجی/آزمون، همان بذرهای تصادفی، و همان قطعات کلاینتی استفاده می‌کنند. قطعات غیرهمسان با نمونه‌گیری دایریکله روی برچسب کلاس‌ها ساخته می‌شوند و \(\alpha \in \{0.01, 0.5, 1.0\}\) برای کنترل ناهمگنی به‌کار می‌رود. مبناهای اصلی شامل آموزش متمرکز، آموزش محلی، \FedAvg{}، \FedProx{}، و \FMRLAVA{} هستند. پیکربندی اجرا از طریق \LatinTerm{Hydra} کنترل می‌شود و شامل تعداد اپیزودهای محلی در هر دور، اندازه حافظه بازپخش، اندازه بچ، تنظیمات \DoubleDQN{}، آموزش مولد، و پرچم‌های واسنجی مجموعه‌باز است.

بخش اعتبارسنجی خارجی به‌صورت مستقل روی \LatinTerm{B-TAT}، \LatinTerm{ToN-IoT}، و \LatinTerm{CIC-IDS2017} اجرا می‌شود. هر مجموعه‌داده نگاشت شناخته‌شده/ناشناخته، تقسیم، و قطعات غیرهمسان ویژه خود را دارد؛ بنابراین این بخش یک مطالعه اعتبارسنجی مستقل است، نه آزمایش انتقال میان‌مجموعه‌داده.

\section{معیارهای ارزیابی}
در ارزیابی مجموعه بسته، باید دقت، \LatinTerm{macro precision}، \LatinTerm{macro recall}، \LatinTerm{macro F1}، \LatinTerm{weighted F1}، دقت متوازن، و ماتریس‌های سردرگمی گزارش شوند. این معیارها نشان می‌دهند که مدل قبل از اعمال ردِ مجموعه‌باز، تا چه حد کلاس‌های شناخته‌شده را درست تفکیک می‌کند.

در ارزیابی مجموعه‌باز، باید \LatinTerm{AUROC}، \LatinTerm{AUPRC}، \LatinTerm{FPR@95\%TPR}، \LatinTerm{TPR@5\%FPR}، \LatinTerm{unknown F1}، نرخ رد ناشناخته، و دقت کلاس‌های شناخته‌شده پس از رد گزارش شوند. این ترکیب لازم است، چون معیارهای مستقل از آستانه و وابسته به آستانه سؤال‌های متفاوتی را پاسخ می‌دهند.

در ارزیابی فدرال، باید دقت نهایی، میانگین 10 دور پایانی، سرعت همگرایی، واریانس بین کلاینت‌ها، نسبت کلاینت‌های منتخب، و هزینه ارتباطی گزارش شوند. کارایی نیز باید با زمان اجرا، بایت‌های منتقل‌شده، و دقت به‌ازای هر مگابایت سنجیده شود.

\section{نتایج و بحث}
نتایج باید بر پایه هشت آزمایش فوق سازمان‌دهی شوند، نه بر مبنای جزئیات پیاده‌سازی. \LatinTerm{E1} باید نشان دهد که مدل اصلی دقت مجموعه بسته را حفظ می‌کند. \LatinTerm{E2} باید ارزش واسنجی \EVT{} را برای رد حملات ناشناخته جدا کند. \LatinTerm{E3} باید \FedAvg{}، \FedProx{}، و \FMRLAVA{} را تحت ناهمگنی فزاینده مقایسه کند. \LatinTerm{E4} باید هم‌زمانیِ فدرال غیرهمسان و رد مجموعه‌باز را بسنجد. \LatinTerm{E5} باید نشان دهد که مدل، وقتی به‌صورت مستقل روی \LatinTerm{B-TAT}، \LatinTerm{ToN-IoT}، و \LatinTerm{CIC-IDS2017} آموزش و ارزیابی می‌شود، همچنان تحت شرایط مجموعه‌باز و غیرهمسان مؤثر باقی می‌ماند. \LatinTerm{E6} باید سهم \EVT{}، مولد، و تجمیع آگاه از سودمندی را جدا کند. \LatinTerm{E7} باید هزینه مکانیزم دو فازه را کمی‌سازی کند. \LatinTerm{E8} باید نشان دهد که جداسازی مجموعه‌باز برای برچسب‌های مختلف سازگار است.

بحث باید صریحاً دادوستدها را تفسیر کند. برای مثال، اگر \FMRLAVA{} پایداری را بهبود دهد اما هماهنگی بیشتری بخواهد، این بده‌بستان باید با نمودار هزینه ارتباطی و نمودار دقت دوربه‌دور نشان داده شود. اگر \EVT{} تشخیص ناشناخته را بهتر کند ولی کمی دقت کلاس‌های شناخته‌شده را نزدیک آستانه کاهش دهد، منحنی کالیبراسیون باید آن را آشکار کند.

\section{حذف مؤلفه و حساسیت}
حذف مؤلفه‌ها باید شامل حذف ردِ \EVT{}، حذف آموزش مولد، جایگزینی \FMRLAVA{} با \FedAvg{}، جایگزینی \FMRLAVA{} با \FedProx{}، و غیرفعال‌سازی انتخاب کلاینت باشد. تحلیل حساسیت نیز باید \(\alpha\) دایریکله، کسر دم \EVT{}، آستانه سودمندی، و بذرهای تصادفی را بررسی کند. این نتایج ارزش هر زیرسامانه و میزان پایداری روش را روشن می‌کنند.

\section{محدودیت‌ها و اعتبارسنجی}
محدودیت‌های اصلی شامل حساسیت کالیبراسیون، هزینه هماهنگی دو مرحله‌ای، و وابستگی به نگاشت برچسبِ ویژه هر مجموعه‌داده در ارزیابی‌های بیرونی است. کیفیت \EVT{} به تقسیم اعتبارسنجی وابسته است و \FMRLAVA{} به وضعیت سمت سرور و پایش اضافی نیاز دارد. این محدودیت‌ها باید صریحاً ذکر شوند تا مقاله بیش از حد کلی‌گرا به نظر نرسد.

\section{بازتولیدپذیری}
مخزن، پیکربندی \LatinTerm{Hydra}، کنترل بذر، ذخیره‌سازی \LatinTerm{checkpoint}، خروجی محلی، و نمودارهای سطح اجرا را فراهم می‌کند. این ویژگی‌ها خط لوله را در مراحل پیش‌پردازش، آموزش، ارزیابی، و تولید شکل‌ها بازتولیدپذیر می‌سازند. نسخه نهایی مقاله باید به خروجی‌های ذخیره‌شده و قرارداد ارزیابی ثابت ارجاع دهد تا نتایج از همان تقسیم و همان فهرست بذرها قابل بازسازی باشند.
